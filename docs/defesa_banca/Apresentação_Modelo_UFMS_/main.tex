% Slide 3: Ciência e Metodologias de Investigação

\begin{frame}{Ciência e Metodologias de Investigação}
    \begin{block}{Definições}
        \begin{itemize}
            \item \textbf{Ciência:} Atividade sistemática de construir explicações para a natureza, utilizando métodos testáveis e reproduzíveis.
            \item \textbf{Estratégia Indutiva:} Busca por padrões em observações para propor explicações *a posteriori*.
            \item \textbf{Estratégia Dedutiva:} Parte de explicações *a priori* para testar predições experimentalmente.
        \end{itemize}
    \end{block}
\end{frame}
